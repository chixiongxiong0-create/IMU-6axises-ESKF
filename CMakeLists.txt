cmake_minimum_required(VERSION 3.20)
project(imu_attitude C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

enable_testing()

add_library(imu_platform_math INTERFACE)
if(UNIX AND NOT APPLE)
  target_link_libraries(imu_platform_math INTERFACE m)
endif()

add_executable(test_types tests/test_types.c)
target_include_directories(test_types PRIVATE include tests)
if(MSVC)
  target_compile_options(test_types PRIVATE /W4 /WX)
else()
  target_compile_options(test_types PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME types COMMAND test_types)

add_executable(test_math tests/test_math.c src/imu_math.c)
target_include_directories(test_math PRIVATE include tests)
target_link_libraries(test_math PRIVATE imu_platform_math)
if(MSVC)
  target_compile_options(test_math PRIVATE /W4 /WX)
else()
  target_compile_options(test_math PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME math COMMAND test_math)

add_executable(test_calibration tests/test_calibration.c src/imu_calibration.c)
target_include_directories(test_calibration PRIVATE include tests)
target_link_libraries(test_calibration PRIVATE imu_platform_math)
if(MSVC)
  target_compile_options(test_calibration PRIVATE /W4 /WX)
else()
  target_compile_options(test_calibration PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME calibration COMMAND test_calibration)

add_executable(test_biquad tests/test_biquad.c src/imu_biquad.c)
target_include_directories(test_biquad PRIVATE include tests)
target_link_libraries(test_biquad PRIVATE imu_platform_math)
if(MSVC)
  target_compile_options(test_biquad PRIVATE /W4 /WX)
else()
  target_compile_options(test_biquad PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME biquad COMMAND test_biquad)

add_executable(test_motion_quality tests/test_motion_quality.c
  src/imu_motion_quality.c)
target_include_directories(test_motion_quality PRIVATE include tests)
target_link_libraries(test_motion_quality PRIVATE imu_platform_math)
if(MSVC)
  target_compile_options(test_motion_quality PRIVATE /W4 /WX)
else()
  target_compile_options(test_motion_quality PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME motion_quality COMMAND test_motion_quality)

add_executable(test_esekf tests/test_esekf.c src/imu_math.c src/imu_esekf.c)
target_include_directories(test_esekf PRIVATE include tests)
target_link_libraries(test_esekf PRIVATE imu_platform_math)
if(MSVC)
  target_compile_options(test_esekf PRIVATE /W4 /WX)
else()
  target_compile_options(test_esekf PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME esekf COMMAND test_esekf)

add_executable(test_complementary tests/test_complementary.c src/imu_math.c
  src/imu_complementary.c)
target_include_directories(test_complementary PRIVATE include tests)
target_link_libraries(test_complementary PRIVATE imu_platform_math)
if(MSVC)
  target_compile_options(test_complementary PRIVATE /W4 /WX)
else()
  target_compile_options(test_complementary PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME complementary COMMAND test_complementary)

add_library(imu STATIC src/imu.c src/imu_math.c src/imu_calibration.c
  src/imu_biquad.c src/imu_motion_quality.c src/imu_esekf.c src/imu_complementary.c)
target_include_directories(imu PUBLIC include)
target_link_libraries(imu PUBLIC imu_platform_math)
add_executable(test_imu tests/test_imu.c)
target_include_directories(test_imu PRIVATE tests)
target_link_libraries(test_imu PRIVATE imu)
if(MSVC)
  target_compile_options(imu PRIVATE /W4 /WX)
  target_compile_options(test_imu PRIVATE /W4 /WX)
else()
  target_compile_options(imu PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(test_imu PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME imu COMMAND test_imu)

add_library(imu_sim STATIC tests/sim/imu_sim.c)
target_include_directories(imu_sim PUBLIC tests include)
target_link_libraries(imu_sim PUBLIC imu)
add_executable(test_scenarios tests/test_scenarios.c tools/imu_compare.c)
target_compile_definitions(test_scenarios PRIVATE IMU_COMPARE_NO_MAIN)
target_link_libraries(test_scenarios PRIVATE imu_sim)
foreach(target imu_sim test_scenarios)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME simulator COMMAND test_scenarios --simulator)
add_test(NAME scenarios COMMAND test_scenarios)
add_test(NAME recovery_metrics COMMAND test_scenarios --recovery)
add_test(NAME comparison COMMAND test_scenarios --comparison)
add_executable(imu_compare tools/imu_compare.c)
add_executable(imu_replay tools/imu_replay.c)
target_link_libraries(imu_replay PRIVATE imu)
set_target_properties(imu_replay PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
add_executable(test_replay tools/imu_replay.c)
target_compile_definitions(test_replay PRIVATE IMU_REPLAY_TEST)
target_include_directories(test_replay PRIVATE tests)
target_link_libraries(test_replay PRIVATE imu)
add_test(NAME replay_parser COMMAND test_replay)
foreach(target imu_replay test_replay)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME replay_smoke COMMAND imu_replay "${CMAKE_SOURCE_DIR}/tests/data/replay_smoke.csv" "${CMAKE_BINARY_DIR}/replay_smoke_output.csv")
target_link_libraries(imu_compare PRIVATE imu_sim)
set_target_properties(imu_compare PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
if(MSVC)
  target_compile_options(imu_compare PRIVATE /W4 /WX)
else()
  target_compile_options(imu_compare PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
